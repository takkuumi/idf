### 1.2 BLE 帧 CRC + 长度校验

```rust
#[test]
fn ble_frame_round_trip_crc() {
    // 构造: tx_id=0x0021, proto_id=0x0894, pdu=modbus_fc10_write
    let frame = build_ble_frame(0x21, 0x0894, &[0x01, 0x10, 0x08, 0x94, 0x00, 0x09, 0x12, /* 18 字节 data */]);
    // CRC 在最后 2 字节, LE
    let calc_crc = modbus_crc16(&frame[..frame.len() - 2]);
    let rx_crc = u16::from_le_bytes([frame[frame.len() - 2], frame[frame.len() - 1]]);
    assert_eq!(calc_crc, rx_crc);
}
```

### 1.3 NVS 持久化 e2e (host test 模拟)

```rust
#[test]
fn write_sn_persists_to_nvs_buffer() {
    let mut nvs = MockNvs::new();
    let cfg = SystemConfig::default();
    cfg.write_reg(0x0894, 0x4142).unwrap(); // SN[0] = "AB"

    // 模拟重启

}
```

## 2. 阶段 2 测试用例 (文本与功能配置)

### 2.1 文本数据布局

```rust
#[test]
fn device_text_count_register_layout() {
    // Android: READ_DEVICE_TEXT_COUNT reg=0x1388
    assert_eq!(regs::DEVICE_TEXT_BASE, 0x1388);
    // count 在 0x1388, 总长在 0x1389, 数据从 0x138A
    assert_eq!(regs::DEVICE_TEXT_BASE, 5000);
    assert_eq!(regs::DEVICE_TEXT_COUNT, 2000);
}

#[test]
fn device_text_utf16_round_trip() {
    let text: Vec<u16> = "你好世界".encode_utf16().collect();
    let mut buf = vec![0u16; 2000];
    for (i, &c) in text.iter().enumerate() {
        buf[i] = c;
    }
    let decoded = String::from_utf16(&buf[..text.len()]).unwrap();
    assert_eq!(decoded, "你好世界");
}
```
